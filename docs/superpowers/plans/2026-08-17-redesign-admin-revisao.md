# Redesenho visual + paginação + edição Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Redesign the 3 existing screens (`/entrar`, shared `Header`, `/admin/revisao`) with shadcn/ui, and add pagination + an edit dialog to `/admin/revisao` so the founder can actually curate the 180 real ENEM questions comfortably.

**Architecture:** shadcn/ui (Base UI primitives under the hood, `components/ui/*` generated by the CLI) layered onto the existing Client Component pages — no change to the underlying auth/data-fetching logic of `/entrar` or `Header`, only their markup. `/admin/revisao` gains URL-based pagination (`?page=N`) and a new `GET /api/admin/topics` endpoint to power an edit dialog that reuses the already-existing `PATCH /api/admin/questions/:id`.

**Tech Stack:** shadcn/ui v4 (`shadcn@4.18.0`, Base UI primitives, not Radix), Tailwind v4 (already configured), Next.js 15 App Router, Drizzle ORM (existing).

## Global Constraints

- TypeScript em todo o código novo
- shadcn/ui instalado só com os componentes usados (`button`, `card`, `input`, `label`, `textarea`, `select`, `dialog`, `pagination`, `badge`), não a biblioteca inteira
- Tema claro fixo, sem dark mode nesta fase
- Cor de destaque (`--primary`, `--ring`): azul, valor exato `oklch(0.546 0.245 262.881)` (Tailwind `blue-600`) — confirmado via `node_modules/tailwindcss/theme.css`
- Paginação fixa em 20 por página (`pageSize` não configurável pelo cliente)
- Sem filtro de área exposto na UI nesta fase (a API já suporta `?area=`, mas a tela de revisão não usa)
- Sem opção de criar assunto/tópico novo pela tela — o dropdown de reclassificação só lista os tópicos já existentes
- Erros exibidos ao usuário nunca expõem detalhe técnico interno (herdado da Fundação)
- ORM com queries parametrizadas — nunca SQL concatenado na mão (herdado)
- Comandos de verificação sempre na ordem `lint → build → typecheck` (build antes de typecheck — `app/layout.tsx` usa `LayoutProps<"/">`, um tipo gerado pelo `next build`; rodar typecheck antes produz um falso erro, incidente real documentado em `docs/superpowers/plans/2026-08-17-login-cadastro.md`)

```typescript
import { and, count, eq } from "drizzle-orm";
import { requireAdmin } from "@/lib/admin";
import { db } from "@/lib/db";
import { questions, topics } from "@/db/schema";

const VALID_STATUSES = ["pendente", "aprovada", "rejeitada"] as const;
const VALID_AREAS = ["linguagens", "humanas", "natureza", "matematica"] as const;
const PAGE_SIZE = 20;

function isValidStatus(value: string): value is (typeof VALID_STATUSES)[number] {
  return VALID_STATUSES.some((s) => s === value);
}

function isValidArea(value: string): value is (typeof VALID_AREAS)[number] {
  return VALID_AREAS.some((a) => a === value);
}

function parsePage(value: string | null): number {
  if (!value || !/^\d+$/.test(value)) {
    return 1;
  }
  const parsed = Number(value);
  return parsed >= 1 ? parsed : 1;
}

export async function GET(request: Request) {
  const session = await requireAdmin(request);
  if (!session) {
    return Response.json({ error: "Não autorizado" }, { status: 401 });
  }

  const url = new URL(request.url);
  const status = url.searchParams.get("status") ?? "pendente";
  const area = url.searchParams.get("area");
  const page = parsePage(url.searchParams.get("page"));

  if (!isValidStatus(status)) {
    return Response.json({ error: "Status inválido" }, { status: 400 });
  }

  if (area && !isValidArea(area)) {
    return Response.json({ error: "Área inválida" }, { status: 400 });
  }

  const conditions = [eq(questions.status, status)];
  if (area) {
    conditions.push(eq(questions.area, area as "linguagens" | "humanas" | "natureza" | "matematica"));
  }

  const [{ value: total }] = await db
    .select({ value: count() })
    .from(questions)
    .where(and(...conditions));

  const rows = await db
    .select({
      id: questions.id,
      year: questions.year,
      examDay: questions.examDay,
      questionNumber: questions.questionNumber,
      area: questions.area,
      statement: questions.statement,
      alternatives: questions.alternatives,
      correctAnswer: questions.correctAnswer,
      imageUrl: questions.imageUrl,
      status: questions.status,
      topic: { id: topics.id, area: topics.area, subject: topics.subject, name: topics.name },
    })
    .from(questions)
    .innerJoin(topics, eq(questions.topicId, topics.id))
    .where(and(...conditions))
    .limit(PAGE_SIZE)
    .offset((page - 1) * PAGE_SIZE);

  return Response.json({ questions: rows, total, page, pageSize: PAGE_SIZE });
}
```

```tsx
"use client";

import { useState, type FormEvent } from "react";
import { useRouter } from "next/navigation";
import { authClient } from "@/lib/auth-client";
import { Button } from "@/components/ui/button";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";

type Mode = "login" | "cadastro";

export default function EntrarPage() {
  const router = useRouter();
  const [mode, setMode] = useState<Mode>("login");
  const [name, setName] = useState("");
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [error, setError] = useState<string | null>(null);
  const [loading, setLoading] = useState(false);

  function redirectByRole(role: string | undefined) {
    router.push(role === "admin" ? "/admin/revisao" : "/");
  }

  async function handleLogin(event: FormEvent) {
    event.preventDefault();
    setError(null);
    setLoading(true);

    try {
      const { data, error: signInError } = await authClient.signIn.email({
        email,
        password,
      });

      if (signInError) {
        setError(signInError.message ?? "Não foi possível entrar.");
        return;
      }

      const role = (data?.user as { role?: string } | undefined)?.role;
      redirectByRole(role);
    } catch {
      setError("Não foi possível conectar. Verifique sua internet e tente novamente.");
    } finally {
      setLoading(false);
    }
  }

  async function handleSignUp(event: FormEvent) {
    event.preventDefault();
    setError(null);
    setLoading(true);

    try {
      const { data, error: signUpError } = await authClient.signUp.email({
        name,
        email,
        password,
      });

      if (signUpError) {
        setError(signUpError.message ?? "Não foi possível criar a conta.");
        return;
      }

      const role = (data?.user as { role?: string } | undefined)?.role;
      redirectByRole(role);
    } catch {
      setError("Não foi possível conectar. Verifique sua internet e tente novamente.");
    } finally {
      setLoading(false);
    }
  }

  return (
    <main className="flex flex-1 items-center justify-center px-4 py-16">
      <Card className="w-full max-w-sm">
        <CardHeader>
          <CardTitle className="text-2xl">
            {mode === "login" ? "Entrar" : "Criar conta"}
          </CardTitle>
        </CardHeader>
        <CardContent>
          <form
            onSubmit={mode === "login" ? handleLogin : handleSignUp}
            className="flex flex-col gap-4"
          >
            {mode === "cadastro" && (
              <div className="flex flex-col gap-2">
                <Label htmlFor="name">Nome</Label>
                <Input
                  id="name"
                  type="text"
                  required
                  value={name}
                  onChange={(event) => setName(event.target.value)}
                />
              </div>
            )}

            <div className="flex flex-col gap-2">
              <Label htmlFor="email">E-mail</Label>
              <Input
                id="email"
                type="email"
                required
                value={email}
                onChange={(event) => setEmail(event.target.value)}
              />
            </div>

            <div className="flex flex-col gap-2">
              <Label htmlFor="password">Senha</Label>
              <Input
                id="password"
                type="password"
                required
                value={password}
                onChange={(event) => setPassword(event.target.value)}
              />
            </div>

            {error && <p className="text-sm text-destructive">{error}</p>}

            <Button type="submit" disabled={loading}>
              {loading ? "Aguarde..." : mode === "login" ? "Entrar" : "Criar conta"}
            </Button>
          </form>

          <Button
            type="button"
            variant="link"
            className="mt-2 px-0"
            onClick={() => {
              setMode(mode === "login" ? "cadastro" : "login");
              setError(null);
            }}
          >
            {mode === "login" ? "Criar conta" : "Já tenho conta — entrar"}
          </Button>
        </CardContent>
      </Card>
    </main>
  );
}
```

```tsx
          </Button>
        </div>
      ) : (

          <Link href="/entrar">Entrar</Link>
        </Button>
```

```tsx
"use client";

import { useEffect, useState, type FormEvent } from "react";
import { Button } from "@/components/ui/button";
import {
  Dialog,
  DialogContent,
  DialogDescription,
  DialogFooter,
  DialogHeader,
  DialogTitle,
} from "@/components/ui/dialog";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from "@/components/ui/select";
import { Textarea } from "@/components/ui/textarea";
import type { Question } from "./page";

type TopicOption = { id: string; area: string; subject: string; name: string };

const LETTERS = ["A", "B", "C", "D", "E"] as const;

export function EditQuestionDialog({
  question,
  onClose,
  onSaved,
}: {
  question: Question;
  onClose: () => void;
  onSaved: (updated: Question) => void;
}) {
  const [statement, setStatement] = useState(question.statement);
  const [alternativeTexts, setAlternativeTexts] = useState<Record<string, string>>(() => {
    const initial: Record<string, string> = {};
    for (const alt of question.alternatives) {
      initial[alt.letter] = alt.text;
    }
    return initial;
  });
  const [correctAnswer, setCorrectAnswer] = useState(question.correctAnswer);
  const [topicId, setTopicId] = useState(question.topic.id);
  const [topics, setTopics] = useState<TopicOption[] | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [saving, setSaving] = useState(false);

  useEffect(() => {
    fetch("/api/admin/topics")
      .then((response) => response.json())
      .then((data) => setTopics(data.topics))
      .catch(() => setError("Não foi possível carregar a lista de assuntos."));
  }, []);

  async function handleSubmit(event: FormEvent) {
    event.preventDefault();
    setError(null);
    setSaving(true);

    try {
      const response = await fetch(`/api/admin/questions/${question.id}`, {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          statement,
          alternatives: LETTERS.map((letter) => ({
            letter,
            text: alternativeTexts[letter] ?? "",
          })),
          correctAnswer,
          topicId,
        }),
      });

      const body = await response.json();

      if (!response.ok) {
        setError(body.error ?? "Não foi possível salvar as alterações.");
        return;
      }

      const updatedTopic = topics?.find((t) => t.id === body.topicId);
      onSaved({
        ...question,
        statement: body.statement,
        alternatives: body.alternatives,
        correctAnswer: body.correctAnswer,
        status: body.status,
        topic: updatedTopic ?? question.topic,
      });
    } catch {
      setError("Não foi possível conectar. Verifique sua internet e tente novamente.");
    } finally {
      setSaving(false);
    }
  }

  return (
    <Dialog open onOpenChange={(open) => !open && onClose()}>
      <DialogContent className="max-h-[85vh] max-w-2xl overflow-y-auto">
        <DialogHeader>
          <DialogTitle>
            Editar questão — {question.year} Dia {question.examDay} Questão{" "}
            {question.questionNumber}
          </DialogTitle>
          <DialogDescription>
            Corrija o enunciado, as alternativas, a resposta correta ou o assunto.
          </DialogDescription>
        </DialogHeader>

        <form onSubmit={handleSubmit} className="flex flex-col gap-4">
          <div className="flex flex-col gap-2">
            <Label htmlFor="statement">Enunciado</Label>
            <Textarea
              id="statement"
              required
              rows={5}
              value={statement}
              onChange={(event) => setStatement(event.target.value)}
            />
          </div>

          {LETTERS.map((letter) => (
            <div key={letter} className="flex flex-col gap-2">
              <Label htmlFor={`alt-${letter}`}>Alternativa {letter}</Label>
              <Input
                id={`alt-${letter}`}
                required
                value={alternativeTexts[letter] ?? ""}
                onChange={(event) =>
                  setAlternativeTexts((prev) => ({ ...prev, [letter]: event.target.value }))
                }
              />
            </div>
          ))}

          <div className="flex flex-col gap-2">
            <Label htmlFor="correctAnswer">Resposta correta</Label>
            <Input
              id="correctAnswer"
              required
              value={correctAnswer}
              onChange={(event) => setCorrectAnswer(event.target.value)}
            />
          </div>

          <div className="flex flex-col gap-2">
            <Label htmlFor="topic">Assunto</Label>
            <Select value={topicId} onValueChange={(value) => setTopicId(value ?? topicId)}>
              <SelectTrigger id="topic" className="w-full">
                <SelectValue placeholder="Escolha um assunto" />
              </SelectTrigger>
              <SelectContent>
                {topics?.map((topic) => (
                  <SelectItem key={topic.id} value={topic.id}>
                    {topic.subject} — {topic.name}
                  </SelectItem>
                ))}
              </SelectContent>
            </Select>
          </div>

          {error && <p className="text-sm text-destructive">{error}</p>}

          <DialogFooter>
            <Button type="button" variant="outline" onClick={onClose}>
              Cancelar
            </Button>
            <Button type="submit" disabled={saving}>
              {saving ? "Salvando..." : "Salvar"}
            </Button>
          </DialogFooter>
        </form>
      </DialogContent>
    </Dialog>
  );
}
```

- [ ] **Step 2: Substituir a página de revisão**

Modify `app/admin/revisao/page.tsx` — substitua o arquivo inteiro por:

```tsx
"use client";

import { Suspense, useEffect, useState } from "react";
import { useSearchParams } from "next/navigation";
import { Badge } from "@/components/ui/badge";
import { Button } from "@/components/ui/button";
import { Card, CardContent, CardHeader } from "@/components/ui/card";
import {
  Pagination,
  PaginationContent,
  PaginationItem,
  PaginationLink,
  PaginationNext,
  PaginationPrevious,
} from "@/components/ui/pagination";
import { EditQuestionDialog } from "./EditQuestionDialog";

export type Alternative = { letter: string; text: string };

export type Topic = { id: string; area: string; subject: string; name: string };

export type Question = {
  id: string;
  year: number;
  examDay: number;
  questionNumber: number;
  area: string;
  statement: string;
  alternatives: Alternative[];
  correctAnswer: string;
  imageUrl: string | null;
  status: string;
  topic: Topic;
};

const PAGE_SIZE = 20;

function RevisaoContent() {
  const searchParams = useSearchParams();
  const page = Math.max(1, Number(searchParams.get("page")) || 1);

  const [questions, setQuestions] = useState<Question[] | null>(null);
  const [total, setTotal] = useState(0);
  const [error, setError] = useState<string | null>(null);
  const [actionError, setActionError] = useState<string | null>(null);
  const [editingQuestion, setEditingQuestion] = useState<Question | null>(null);

  useEffect(() => {
    setQuestions(null);
    fetch(`/api/admin/questions?status=pendente&page=${page}`)
      .then((response) => {
        if (response.status === 401) {
          throw new Error("Não autorizado — faça login com uma conta admin.");
        }
        if (!response.ok) {
          throw new Error("Erro ao carregar questões. Tente novamente.");
        }
        return response.json();
      })
      .then((data) => {
        setQuestions(data.questions);
        setTotal(data.total);
      })
      .catch((err: Error) => setError(err.message));
  }, [page]);

  async function handleAction(id: string, status: "aprovada" | "rejeitada") {
    const response = await fetch(`/api/admin/questions/${id}`, {
      method: "PATCH",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ status }),
    });
    if (response.ok) {
      setQuestions((prev) => (prev ? prev.filter((q) => q.id !== id) : prev));
      setTotal((prev) => prev - 1);
      setActionError(null);
    } else {
      setActionError("Erro ao processar ação. Tente novamente.");
    }
  }

  function handleEdited(updated: Question) {
    setQuestions((prev) =>
      prev ? prev.map((q) => (q.id === updated.id ? updated : q)) : prev
    );
    setEditingQuestion(null);
  }

  if (error) {
    return <p className="p-8 text-destructive">{error}</p>;
  }

  if (!questions) {
    return <p className="p-8">Carregando...</p>;
  }

  const totalPages = Math.max(1, Math.ceil(total / PAGE_SIZE));

  return (
    <main className="mx-auto flex max-w-3xl flex-col gap-6 px-4 py-8">
      <h1 className="text-2xl font-semibold">Revisão de questões ({total} pendentes)</h1>

      {actionError && <p className="text-sm text-destructive">{actionError}</p>}

      {questions.length === 0 && (
        <p className="text-muted-foreground">Nenhuma questão pendente de revisão.</p>
      )}

      <div className="flex flex-col gap-4">
        {questions.map((question) => (
          <Card key={question.id}>
            <CardHeader className="flex flex-row items-center justify-between gap-2">
              <div>
                <p className="font-semibold">
                  {question.year} — Dia {question.examDay} — Questão {question.questionNumber}
                </p>
                <p className="text-sm text-muted-foreground">
                  {question.topic.subject} — {question.topic.name}
                </p>
              </div>
              <Badge variant="secondary">{question.status}</Badge>
            </CardHeader>
            <CardContent className="flex flex-col gap-4">
              <p>{question.statement}</p>
              {question.imageUrl && (
                // eslint-disable-next-line @next/next/no-img-element
                <img
                  src={question.imageUrl}
                  alt="Figura da questão"
                  className="max-w-sm rounded border"
                />
              )}
              <ul className="flex flex-col gap-1">
                {question.alternatives.map((alt) => (
                  <li
                    key={alt.letter}
                    className={
                      alt.letter === question.correctAnswer
                        ? "font-medium text-primary"
                        : undefined
                    }
                  >
                    {alt.letter}) {alt.text}
                    {alt.letter === question.correctAnswer && " ✓"}
                  </li>
                ))}
              </ul>
              <div className="flex gap-2">
                <Button onClick={() => handleAction(question.id, "aprovada")}>Aprovar</Button>
                <Button
                  variant="destructive"
                  onClick={() => handleAction(question.id, "rejeitada")}
                >
                  Rejeitar
                </Button>
                <Button variant="outline" onClick={() => setEditingQuestion(question)}>
                  Editar
                </Button>
              </div>
            </CardContent>
          </Card>
        ))}
      </div>

      {totalPages > 1 && (
        <Pagination>
          <PaginationContent>
            <PaginationItem>
              <PaginationPrevious
                href={`/admin/revisao?page=${Math.max(1, page - 1)}`}
                aria-disabled={page === 1}
                className={page === 1 ? "pointer-events-none opacity-50" : undefined}
              />
            </PaginationItem>
            {Array.from({ length: totalPages }, (_, index) => index + 1).map((pageNumber) => (
              <PaginationItem key={pageNumber}>
                <PaginationLink
                  href={`/admin/revisao?page=${pageNumber}`}
                  isActive={pageNumber === page}
                >
                  {pageNumber}
                </PaginationLink>
              </PaginationItem>
            ))}
            <PaginationItem>
              <PaginationNext
                href={`/admin/revisao?page=${Math.min(totalPages, page + 1)}`}
                aria-disabled={page === totalPages}
                className={
                  page === totalPages ? "pointer-events-none opacity-50" : undefined
                }
              />
            </PaginationItem>
          </PaginationContent>
        </Pagination>
      )}

      {editingQuestion && (
        <EditQuestionDialog
          question={editingQuestion}
          onClose={() => setEditingQuestion(null)}
          onSaved={handleEdited}
        />
      )}
    </main>
  );
}

export default function RevisaoPage() {
  return (
    <Suspense fallback={<p className="p-8">Carregando...</p>}>
      <RevisaoContent />
    </Suspense>
  );
}
```
